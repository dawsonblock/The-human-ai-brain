### Key Tests

#### PSD Clamp Test
```cpp
TEST(QuantumCore, PSDClamp) {
    QuantumWorkspace qw(7);
    qw.evolve(0.01);
    
    Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> solver(qw.rho());
    EXPECT_GE(solver.eigenvalues().minCoeff(), -1e-9);
    EXPECT_NEAR(qw.rho().trace(), 1.0, 1e-9);
}
```

#### Entropy Drift Test
```cpp
TEST(QuantumCore, EntropyMonotonic) {
    QuantumWorkspace qw(7);
    double prev_entropy = qw.entropy();
    
    for (int i = 0; i < 100; ++i) {
        qw.evolve(0.01);
        double curr_entropy = qw.entropy();
        EXPECT_GE(curr_entropy, prev_entropy - 1e-6);
        prev_entropy = curr_entropy;
    }
}
```

#### Collapse Timing Test
```cpp
TEST(QuantumCore, CollapseTiming) {
    QuantumWorkspace qw(7);
    auto start = std::chrono::steady_clock::now();
    
    while (!qw.should_collapse()) {
        qw.evolve(0.01);
    }
    
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(
        std::chrono::steady_clock::now() - start
    );
    
    EXPECT_LE(duration.count(), 120);  // max_dwell_ms
}
```

## 🐳 Docker & Kubernetes

### Multi-Stage Dockerfile
```dockerfile
# Builder stage with all dependencies
FROM ubuntu:22.04 AS builder
RUN apt-get update && apt-get install -y \
    cmake g++ libeigen3-dev libgrpc++-dev \
    libssl-dev libfaiss-dev libprometheus-cpp-dev
COPY . /src
RUN cmake -B /build -S /src -DCMAKE_BUILD_TYPE=Release && \
    cmake --build /build -j$(nproc)

# Runtime stage - minimal
FROM ubuntu:22.04
RUN apt-get update && apt-get install -y \
    libstdc++6 libssl3 libgrpc++1.51 && \
    apt-get clean && rm -rf /var/lib/apt/lists/*
    
COPY --from=builder /build/brain_server /usr/local/bin/
COPY --from=builder /usr/local/bin/grpc_health_probe /usr/local/bin/
COPY configs/quantum.yaml /etc/brain/quantum.yaml

USER 65534:65534
EXPOSE 50051 9090
ENTRYPOINT ["/usr/local/bin/brain_server"]
CMD ["--config", "/etc/brain/quantum.yaml"]
```

## 🔄 CI/CD Pipeline

### GitHub Actions Workflow
```yaml
name: Production Build v3.4.0

on: [push, pull_request]

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      - name: Install dependencies
        run: |
          sudo apt-get update
          sudo apt-get install -y cmake g++ libeigen3-dev \
            libgrpc++-dev libssl-dev libfaiss-dev
      
      - name: Configure CMake
        run: cmake -B build -DCMAKE_BUILD_TYPE=Release -DBUILD_TESTS=ON
      
      - name: Build
        run: cmake --build build -j$(nproc)
      
      - name: Run tests
        run: cd build && ctest --output-on-failure
      
      - name: Benchmark
        run: ./build/tools/bench_entropy
      
      - name: README compliance check
        run: python3 scripts/check_readme_compliance.py
  
  security-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      - name: Build Docker image
        run: docker build -t brain-ai:latest .
      
      - name: Trivy scan
        uses: aquasecurity/trivy-action@master
        with:
          image-ref: 'brain-ai:latest'
          format: 'sarif'
          output: 'trivy-results.sarif'
      
      - name: Generate SBOM
        run: syft brain-ai:latest -o spdx-json > sbom.json
  
  deploy:
    needs: [build-and-test, security-scan]
    runs-on: ubuntu-latest
    if: github.ref == 'refs/heads/main'
    steps:
      - name: Deploy to staging
        run: kubectl apply -f k8s/
```
